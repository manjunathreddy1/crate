import * as React from "react";
import * as ReactDom from "react-dom";
import { Version } from "@microsoft/sp-core-library";
import {
  type IPropertyPaneConfiguration,
  PropertyPaneTextField,
} from "@microsoft/sp-property-pane";
import { BaseClientSideWebPart } from "@microsoft/sp-webpart-base";
import { IReadonlyTheme } from "@microsoft/sp-component-base";

import * as strings from "AllEventsWebPartStrings";
import AllEvents from "./components/AllEvents";
import { IAllEventsProps } from "./components/IAllEventsProps";
import { MSGraphClientV3 } from "@microsoft/sp-http";

export interface IAllEventsWebPartProps {
  description: string;
}

export default class AllEventsWebPart extends BaseClientSideWebPart<IAllEventsWebPartProps> {
  private _isDarkTheme: boolean = false;
  private _environmentMessage: string = "";

  private searchListItems(): void {
    const currentUser = this.context.pageContext.user;
    console.log(currentUser);
    this.context.msGraphClientFactory
      .getClient("3")
      .then((client: MSGraphClientV3) => {
        const requestBody = {
          requests: [
            {
              entityTypes: ["listItem"],
              query: {
                queryString:
                  "{searchTerms} AND ContentTypeId:0x0101009D1CB255DA76424F860D91F20E6C4118002BDEE93B36EEED43ACAC4AB6671FE07900BE215E497B31BE4B81F5B5747FDB8265 (NormSiteID:4443d6d5-432c-4a38-9de6-a0a6421b7b41) (ModernAudienceAadObjectIds:{User.Audiences} OR NOT IsAudienceTargeted:true)",
              },
              fields: [
                "Title",
                "AuthorByline",
                "PictureThumbnailURL",
                "AuthorOWSUSER",
                "ows_StartDate",
                "StartDateOWSDATE",
                "EndDateOWSTDATE",
                "Audiences",
                "ModernAudienceAadObjectIds",
              ],
            },
          ],
        };

        return client.api("/search/query").version("v1.0").post(requestBody);
      })
      .then((response) => {
        const results = response?.value?.[0]?.hitsContainers?.[0]?.hits || [];
        let html = `<div><strong>Found ${results.length} results</strong><ul>`;
        for (const hit of results) {
          const title = hit.resource?.fields?.Title || "(no title)";
          html += `<li>${title}</li>`;
        }
        html += `</ul></div>`;
        this.domElement.innerHTML = html;
      })
      .catch((error) => {
        console.error("Graph search failed:", error);
        this.domElement.innerHTML = `<div>Error: ${error.message}</div>`;
      });
  }

  public render(): void {
    this.searchListItems();

    const element: React.ReactElement<IAllEventsProps> = React.createElement(
      AllEvents,
      {
        description: this.properties.description,
        isDarkTheme: this._isDarkTheme,
        environmentMessage: this._environmentMessage,
        hasTeamsContext: !!this.context.sdks.microsoftTeams,
        userDisplayName: this.context.pageContext.user.displayName,
      }
    );

    ReactDom.render(element, this.domElement);
  }

  protected onInit(): Promise<void> {
    return this._getEnvironmentMessage().then((message) => {
      this._environmentMessage = message;
    });
  }

  private _getEnvironmentMessage(): Promise<string> {
    if (!!this.context.sdks.microsoftTeams) {
      // running in Teams, office.com or Outlook
      return this.context.sdks.microsoftTeams.teamsJs.app
        .getContext()
        .then((context) => {
          let environmentMessage: string = "";
          switch (context.app.host.name) {
            case "Office": // running in Office
              environmentMessage = this.context.isServedFromLocalhost
                ? strings.AppLocalEnvironmentOffice
                : strings.AppOfficeEnvironment;
              break;
            case "Outlook": // running in Outlook
              environmentMessage = this.context.isServedFromLocalhost
                ? strings.AppLocalEnvironmentOutlook
                : strings.AppOutlookEnvironment;
              break;
            case "Teams": // running in Teams
            case "TeamsModern":
              environmentMessage = this.context.isServedFromLocalhost
                ? strings.AppLocalEnvironmentTeams
                : strings.AppTeamsTabEnvironment;
              break;
            default:
              environmentMessage = strings.UnknownEnvironment;
          }

          return environmentMessage;
        });
    }

    return Promise.resolve(
      this.context.isServedFromLocalhost
        ? strings.AppLocalEnvironmentSharePoint
        : strings.AppSharePointEnvironment
    );
  }

  protected onThemeChanged(currentTheme: IReadonlyTheme | undefined): void {
    if (!currentTheme) {
      return;
    }

    this._isDarkTheme = !!currentTheme.isInverted;
    const { semanticColors } = currentTheme;

    if (semanticColors) {
      this.domElement.style.setProperty(
        "--bodyText",
        semanticColors.bodyText || null
      );
      this.domElement.style.setProperty("--link", semanticColors.link || null);
      this.domElement.style.setProperty(
        "--linkHovered",
        semanticColors.linkHovered || null
      );
    }
  }

  protected onDispose(): void {
    ReactDom.unmountComponentAtNode(this.domElement);
  }

  protected get dataVersion(): Version {
    return Version.parse("1.0");
  }

  protected getPropertyPaneConfiguration(): IPropertyPaneConfiguration {
    return {
      pages: [
        {
          header: {
            description: strings.PropertyPaneDescription,
          },
          groups: [
            {
              groupName: strings.BasicGroupName,
              groupFields: [
                PropertyPaneTextField("description", {
                  label: strings.DescriptionFieldLabel,
                }),
              ],
            },
          ],
        },
      ],
    };
  }
}
